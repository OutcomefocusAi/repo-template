```js
// Offline (no network, no token): release-please 17.6.0. Manifest.fromManifest reads the repo's
// release-please-config.json and .release-please-manifest.json through a stub GitHub, then the
// strategy built from that config gives the release pull request branch and title, and, for the
// merged release pull request, the tag and GitHub release name.
// Usage: node tag-probe.js <dir with release-please-config.json> <manifest json file> <release PR body file>
const fs = require('fs');
const { Manifest } = require('release-please/build/src/manifest');
const { buildStrategy } = require('release-please/build/src/factory');
const { parseConventionalCommits } = require('release-please/build/src/commit');
const [dir, manifestFile, bodyFile] = process.argv.slice(2);
const files = { 'release-please-config.json': dir + '/release-please-config.json', '.release-please-manifest.json': manifestFile };
const github = { repository: { owner: 'OutcomefocusAi', repo: 'repo-template', defaultBranch: 'main' },
  getFileJson: async p => JSON.parse(fs.readFileSync(files[p], 'utf8')) };
(async () => {
  const m = await Manifest.fromManifest(github, 'main');
  const cfg = m.repositoryConfig['.'];
  const s = await buildStrategy({ github, ...cfg, path: '.', targetBranch: 'main' });
  const commits = parseConventionalCommits([{ sha: '56975b7', message: 'feat: bootstrap repo-template (#1)', files: [] }]);
  const pr = await s.buildReleasePullRequest(commits, undefined);
  const rel = await s.buildRelease({ title: pr.title.toString(), body: fs.readFileSync(bodyFile, 'utf8'),
    headBranchName: pr.headRefName, baseBranchName: 'main', number: 11, labels: [], files: [], sha: 'mergesha' });
  console.log(`config: package-name=${cfg.packageName} include-v-in-tag=${cfg.includeVInTag} include-component-in-tag=${cfg.includeComponentInTag}`);
  console.log(`  release PR branch: ${pr.headRefName}`);
  console.log(`  release PR title:  ${pr.title.toString()}`);
  console.log(`  tag on merge:      ${rel && rel.tag.toString()}`);
  console.log(`  release name:      ${rel && rel.name}`);
})().catch(e => { console.error('ERR', e.stack); process.exit(1); });
```
